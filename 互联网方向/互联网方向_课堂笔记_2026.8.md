## core /base_page.py

```python
from core.driver_manager import *  # dr, wait_10

class BasePage:
    def __init__(self, driver=dr):
        self.driver = driver
        self.wait = wait_10

    def open(self, url):
        self.driver.get(url)

    def find_element(self, locator):
        return self.wait.until(EC.presence_of_element_located(locator))

    def input_text(self, locator, text):
        self.find_element(locator).send_keys(text)

    def click(self, locator):
        self.find_element(locator).click()
```

```js
                function (err, res) {
                    if (err) {  // 异常处理：网络错误或请求超时
```

## Postman中生成签名

> 返回：前置、后置操作（完整进阶任务）

```js
///// 场景二：动态生成签名，并设置到 Authorization头
// 工作流程：设置请求参数 -> 取密钥 -> 组装数据 -> 算签名 -> 塞入 Header
// 1. 直接设置好输入参数值
params = pm.request.body.urlencoded
params.upsert({ key: "param1", value: ??? });
params.upsert({ key: "param2", value: ??? });

// 2. 设定签名密钥（问开发要）
secretKey = 'your_secret_key';

// 3. 生成当前时间戳 》获取最新的请求体 》拼接上当前时间戳（拼接的先后顺序以及连接符，也需要问开发），生成待签名字符串
timestamp = Date.now();
// 说明：.toJSON():将 body先转为 js对象，.stringify(): 再将 js对象转为 JSON字符串
requestBody = JSON.stringify(pm.request.body.toJSON());
signData = requestBody + '&timestamp=' + timestamp;

// 4. 使用 Postman 内置的 CryptoJS 库计算 HMAC-SHA256 签名（与开发确认使用的签名算法，还可以是MD5 或RSA）
sign = CryptoJS.HmacSHA256(signData, secretKey).toString(CryptoJS.enc.Hex);

// 5. 将签名直接添加到请求头（使用 upsert 避免重复添加）
params = pm.request.headers
params.upsert({key: 'Authorization',  value: 'Bearer ' + sign});
```

### 基础层: 对 requests库进行封装

处理所有与 HTTP 协议相关的底层细节

#### core /utils /client.py

```python
import requests

class HttpClient:
    # HTTP 客户端封装类，负责处理所有底层的网络请求。

    def __init__(self, base_url):
        # 1. 保存基础 URL，例如：https://api.example.com
        # 在调用具体接口时，只需传入路径部分即可，如 /api/v1/login
        self.base_url = base_url

        # 2. 创建一个 Session 对象
        # 自动管理和持久化 Cookies：
        #     用户登录后，服务器返回的 sessionid 会被 Session 自动保存，
        #     后续的请求（如查询购物车）会自动带上这个 Cookie，无需手动处理。
        self.session = requests.Session()

    def post(self, api_path, params=None):
        """
        封装 POST 请求方法。
        :param api_path: 接口的路径，例如 "/api/v1/login"
        :param params: 请求体中的 JSON 数据，通常是一个字典
        :return: requests.Response 对象
        """
        # 1. 拼接完整的 URL。例如：url = "https://api.example.com/api/v1/login"
        url = f"{self.base_url}{api_path}"

        try:
            # 2. 发送 POST 请求
            # 使用 self.session.post 代替原生的 requests.post，
            #       以利用 Session 的 Cookie 管理功能
            # json=params 会自动将字典序列化为 JSON 字符串，并设
            #       置请求头 Content-Type 为 application/json
            response = self.session.post(url, json=params)

            # 3. 返回原始的响应对象（由上层决定如何解析（如 .json() 或 .text）
            return response
        except Exception as e:
            # 4. 异常处理：可以在这里统一记录，然后重新抛出异常，让测试用例捕获
            print(f"请求发送失败: {e}")
            raise
```

### 接口层：API Object

```python
from core.utils.client import HttpClient

class UserAPIs:
    """
用户相关的接口集合类：
一个业务模块一个类
2.（业务模块中的）一个api一个类中的方法函数
    将登录、注册、获取用户信息等接口封装在一起。
    """
    def __init__(self, client):
        # UserAPIs 只专注于业务及请求参数，不需要关心网络请求 client的细节

        """ http://api.example.com/api/v1/login
        :return: 返回底层的响应对象 (requests.Response)
        """
        # 准备请求参数 > 发送请求
        payload = {"username": uname, "password": pwd}
        return self.client.post("/api/v1/login", params=payload)

    def register(self, uname, pwd, age, sex):
```

## Python中私有属性的读/写方案

私有属性（private）：外部（主程序，该类的继承子类），均不可以读取和修改（系统会报错）

- 私有属性的“读取 和 修改”（getter/ setter方法）：
  通过专属的装饰器，将“方法伪装成（公有）属性”
  - @property / @属性.setter 分别添加私有属性的“读取 / 修改”方法
      —— 注意：@属性.setter不能单独添加，必须搭配 @property 使用
      主程序中，可以像访问公有属性一样“读取 或 修改”类中的私有属性，并且可以添加必要的前置审核

```python
class 类A:
    def __init__(self, ...):
        self.__(私有)属性X = ...		# 注意：必须以 双 _ 开头
        self.__(私有)属性Y = ...

    #### 提供对 __私有属性X的 "读 / 写" 功能
    @property
    def (私有)属性X(self):			# 注意：可以移除双下划线！！！
        ...         					# 可增加额外的前置逻辑
        return self.__私有属性X
    @属性X.setter
    def (私有)属性X(self, 形参X):		# 注意：可以移除双下划线！！！
        ...         					# 可增加对形参X的验证逻辑
        self.__私有属性X = 形参X

(对象)变量a = 类A
# 注意：与方法定义体保持一下，即可以移除双下划线！！！
(对象)变量a.(私有)属性X = 值x     # 自动触发 (对象)变量a.@属性X.setter。
(对象)变量a.(私有)属性X           # 自动触发 (对象)变量a.@属性X.getter
```

```python
class Ren:
    def __init__(self, mz, nl):
        self.name = mz      # 公有属性
        self.__age = nl     # 私有属性

    @property               # （私有）属性的 读取 方法
    def age(self):
        return 18
    @age.setter
    def age(self, nl2):     # （私有）属性的 修改 方法
        self.__age = nl2

    def get_real_age(self, pwd):
        if pwd == 12345:    # 如果 口令正确，则返回真实年龄
            return self.__age
        else:
            return self.age        # 自动触发 self.age(self) 的调用

r1 = Ren(mz='乔碧萝', nl=50)
# print(r1.name, r1.__age)    # 报错：私有属性，外界是无法读取或是修改
print(r1.name, r1.age)      # r1.age 自动触发 r1.age(self) 的调用：乔碧萝   18
# r1.__age = 51               # 报错
r1.age = 51                 # 自动触发：r1.age(nl2 = 51)
print(r1.age)               # 18

a = r1.get_real_age(pwd=12345)      # 乔碧萝实际年龄是 51
print(f'{r1.name}实际年龄是{a}')
```
