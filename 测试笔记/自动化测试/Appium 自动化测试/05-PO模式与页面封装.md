## 一、为什么需要 PO（移动端更痛）

脚本式写法的问题在移动端被放大：

```python title="testcases/test_xxx.py"
# ❌ 反例：定位 + 操作 + 断言混在一起，考研帮 UI 一改全崩
def test_login():
    driver.find_element(AppiumBy.ID, "com.tal.kaoyan:id/kylogin_phone_input_phonenum").send_keys("138...")
    driver.find_element(AppiumBy.ID, "com.tal.kaoyan:id/kylogin_phone_input_code").send_keys("123456")
    driver.find_element(AppiumBy.ID, "com.tal.kaoyan:id/loginCodeLoginBtn").click()
    assert ...
```

> **Page Object：把每个页面抽象成类，元素定位 + 操作封装在类里，用例只调方法、做断言。** UI 变了只改页面类。

---

## 二、PO 分层结构

```text
testcases/          测试用例层：只关心业务流程和断言
   ↑ 调用
pages/              页面对象层：封装元素定位 + 页面操作
   ↑ 继承
pages/base_page.py  基类：所有页面的公共方法（点击、输入、等待、截图）
   ↑ 依赖
utils/              工具层：driver 工厂、等待、日志、截图、配置
```

## 三、BasePage（Appium 版）

```python title="pages/base_page.py"
"""
BasePage：所有页面对象的基类
位置：pages/base_page.py
"""
from appium.webdriver.common.appiumby import AppiumBy
from utils.wait_helper import WaitHelper
import time


class BasePage:
    def __init__(self, driver):
        self.driver = driver
        self.wait = WaitHelper(driver, timeout=10)

    # ---------- 基础操作 ----------
    def find(self, locator, timeout=None):
        return self.wait.element_visible(locator, timeout)

    def finds(self, locator, timeout=None):
        return self.driver.find_elements(*locator)

    def click(self, locator, timeout=None):
        self.wait.element_clickable(locator, timeout).click()
        return self

    def type(self, locator, text, timeout=None):
        el = self.wait.element_visible(locator, timeout)
        el.clear()
        el.send_keys(text)
        return self

    def get_text(self, locator, timeout=None):
        return self.find(locator, timeout).text

    def is_element_visible(self, locator, timeout=3) -> bool:
        return self.wait.is_element_visible(locator, timeout)

    # ---------- 移动端常用 ----------
    def swipe_up(self, duration=800):
        size = self.driver.get_window_size()
        w, h = size["width"], size["height"]
        self.driver.swipe(w // 2, int(h * 0.8), w // 2, int(h * 0.2), duration)
        return self

    def back(self):
        self.driver.back()
        return self

    def screenshot(self, name=None):
        import os
        name = name or f"screenshot_{int(time.time())}"
        os.makedirs("screenshots", exist_ok=True)
        path = f"screenshots/{name}.png"
        self.driver.save_screenshot(path)
        return path
```

## 四、业务页面对象

```python title="pages/login_page.py"
from appium.webdriver.common.appiumby import AppiumBy

from pages.base_page import BasePage


class LoginPage(BasePage):
    # 手机号
    phone_input = (
        AppiumBy.ID,
        "com.tal.kaoyan:id/kylogin_phone_input_phonenum"
    )

    # 验证码
    code_input = (
        AppiumBy.ID,
        "com.tal.kaoyan:id/kylogin_phone_input_code"
    )

    # 获取验证码
    get_code_btn = (
        AppiumBy.ID,
        "com.tal.kaoyan:id/kylogin_phone_input_codeget"
    )

    # 登录按钮
    login_btn = (
        AppiumBy.ID,
        "com.tal.kaoyan:id/loginCodeLoginBtn"
    )

    # 用户协议
    agreement_cb = (
        AppiumBy.ID,
        "com.tal.kaoyan:id/loginTreatyCheckboxPhone"
    )

    def load(self):
        self.wait.element_visible(
            self.phone_input,
            timeout=15
        )
        return self

    def agree(self):
        checkbox = self.find(self.agreement_cb)

        if not checkbox.is_selected():
            checkbox.click()

        return self

    def login(self, phone, code):
        self.type(
            self.phone_input,
            phone
        )

        self.type(
            self.code_input,
            code
        )

        self.agree()

        self.click(
            self.login_btn
        )

        return self

    # ---------- 分步操作（等待/数据驱动用例复用）----------
    def fill_phone(self, phone):
        """只填手机号"""
        return self.type(self.phone_input, phone)

    def fill_code(self, code):
        """只填验证码"""
        return self.type(self.code_input, code)

    def agree_protocol(self):
        """勾选用户协议"""
        return self.agree()

    def get_error_message(self, timeout: int = 5) -> str:
        """登录失败时获取错误提示文案（Toast）。

        已实测：考研帮登录失败提示以 Toast 形式弹出，
        文案未知时用 toast_any 抓任意 Toast 兜底；没弹则返回空串。
        """
        try:
            toast = self.wait.toast_any(timeout)
            return toast.text or ""
        except Exception:
            return ""

    def is_login_btn_enabled(self, timeout: int = 2) -> bool:
        """登录按钮当前是否可点击。

        已实测：手机号/验证码为空时，App 前置校验会把登录按钮灰置
        （不弹任何提示），此时 element_to_be_clickable 会超时。
        """
        try:
            self.wait.element_clickable(self.login_btn, timeout)
            return True
        except Exception:
            return False

```

from appium.webdriver.common.appiumby import AppiumBy
from pages.base_page import BasePage


class HomePage(BasePage):

    # 首页 Tab
    home_tab = (
        AppiumBy.ID,
        "com.tal.kaoyan:id/main_button_home"
    )

    # 我的 Tab
    myself_tab = (
        AppiumBy.ID,
        "com.tal.kaoyan:id/main_button_myself"
    )

    # 课程 Tab
    course_tab = (
        AppiumBy.ID,
        "com.tal.kaoyan:id/main_button_course"
    )

    # 学习 Tab
    study_tab = (
        AppiumBy.ID,
        "com.tal.kaoyan:id/main_button_study"
    )


    def is_loaded(self):
        """
        判断首页是否加载完成
        """
        return self.is_element_visible(
            self.home_tab,
            timeout=10
        )


    def click_home(self):
        self.click(self.home_tab)
        return self


    def click_myself(self):
        self.click(self.myself_tab)
        return self

## 五、测试用例（只剩流程 + 断言）

```python title="testcases/test_login_po.py"
"""
登录用例（PO 版）
位置：testcases/test_login_po.py

注意：考研帮登录是"手机号 + 短信验证码"（不是密码）。
"登录成功跳首页"需要真实验证码（短信平台/测试环境），自动化受外部依赖限制；
可自动验证的是"验证码错误"场景（留在登录页）。
"""
from pages.home_page import HomePage
from pages.login_page import LoginPage


class TestLoginPo:
    def test_login_wrong_code(self, driver):
        """验证码错误：登录失败，停留在登录页（不跳首页）"""
        login = LoginPage(driver).load()
        login.login("17867940894", "wrongpass")

        # 稳定断言：错误验证码不跳转首页
        assert not HomePage(driver).is_loaded(), "验证码错误不应跳转首页"
        # 可选增强：抓到 Toast 则打印文案（uiautomator2 事件流不稳定，不硬断言）
        print(f"  错误提示: {login.get_error_message()!r}")

    def test_login_btn_disabled_when_empty(self, driver):
        """手机号/验证码为空：登录按钮灰置不可点（不弹提示）"""
        login = LoginPage(driver).load()
        login.fill_phone("")
        login.fill_code("")
        login.agree_protocol()
        assert not login.is_login_btn_enabled(), "输入为空时登录按钮应不可点击"
```

> [!note]
>
> `driver` 从哪来？
> 由 `conftest.py` 的 `@pytest.fixture` 注入（函数参数），见 [[01-环境搭建与工程初始化#3.7 pytest 配置 pytest.ini + 夹具 conftest.py]] 与 `03-实战项目/conftest.py`。
> 不要在 `setup_method` 里用 `self.driver`——pytest 的 `driver` 夹具是参数注入的，直接作为用例参数即可。

> [!warning]
>
> 文档旧版用例里的两个坑（已实测修正）：
> - `EC.activity_started(...)`：Python 的 selenium `expected_conditions` **没有**这个方法，会 `AttributeError`，改用 `driver.wait_activity` 或等首页元素（见 [[03-等待机制#3.1 等待 Activity 切换（App 专属）]]）。
> - `home.get_welcome_text()`：`HomePage` 未定义此方法，会 `AttributeError`；首页断言用 `HomePage(driver).is_loaded()` 即可。

## 六、PO 模式好处（移动端视角）

| 维度 | 脚本式 | PO 模式 |
| --- | --- | --- |
| 元素定位 | 散落各处 | 集中在页面类顶部 |
| UI 变更（考研帮版本升级） | 改所有用例 | 只改页面类 |
| 复用 | 登录逻辑重寫 | 所有用例调 `login()` |
| 多设备 | 难维护 | 页面/用例分层开发 |
| 可读性 | 流程+定位+断言混杂 | 用例只剩业务流程 |

---

## 七、PO 最佳实践

1. **Page 里不写断言**：断言属于用例，Page 只提供数据/操作
2. **方法返回 self**：支持链式 `page.load().login()`
3. **定位器集中声明**：放类顶部，命名清晰
4. **页面跳转返回目标页对象**：`login()` 成功后可 `return HomePage(self.driver)`
5. **公共方法提 BasePage**：`click/type/swipe_up/back` 都放基类
6. **页面粒度适中**：一个页一个类
7. **配合 fixture 管理 driver**：生命周期交给 pytest
